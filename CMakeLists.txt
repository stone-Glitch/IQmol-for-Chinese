#-------------------------------------------------------------------
# QGLViewer / yaml-cpp 静态链接宏 (Windows 必需):
# 二者均为静态库 (modules/CMakeLists.txt 用 add_library(QGLViewer STATIC),
# yaml-cpp 继承 BUILD_SHARED_LIBS=OFF), 但它们的头文件靠编译宏判断"静态还是 DLL":
#   libQGLViewer/QGLViewer/config.h: 未定义 QGLVIEWER_STATIC 时 -> __declspec(dllimport)
#   yaml-cpp/include/yaml-cpp/dll.h : 未定义 YAML_CPP_STATIC_DEFINE 时 -> dllimport
# 而 BUILD_SHARED_LIBS=OFF 只影响库本体如何编译, 不会传递到消费者头文件。
# 结果: 所有引用这些头的 .o 生成 __imp_ 导入符号, 但静态库里只有普通符号,
# 链接 IQmol.exe 时报大量 undefined reference to `__imp__ZN9QGLViewer.../__imp__ZN4YAML...
# 故在此显式定义"静态库"宏, 使两类头文件均按静态语义展开(宏为空/无 dllimport)。
#-------------------------------------------------------------------

if (LINUX)
   find_package(OpenBabel3 REQUIRED)
   include_directories(/usr/include/openbabel3)
else()
   include_directories(modules/openbabel/include)
   include_directories(${CMAKE_CURRENT_BINARY_DIR}/modules/openbabel/include)
   # babelconfig.h 是 OpenBabel 在 configure 期由 src/config.h.cmake 生成的头文件，
   # 嵌入构建时偶发未生成，会导致 elements.h 等包含 <openbabel/babelconfig.h> 失败。
   # 这里先用已知 3.1.1 取值显式生成一份兜底；若 OpenBabel 自身 configure_file 随后运行，
   # 会用其正确版本覆盖本文件。
   set(_OB_DATADIR "${CMAKE_CURRENT_SOURCE_DIR}/modules/openbabel/data")
   set(BABEL_DATADIR "${_OB_DATADIR}")
   set(BABEL_MAJ_VER 3)
   set(BABEL_MIN_VER 1)
   set(BABEL_PATCH_VER 1)
   set(BABEL_VERSION "3.1.1")
   set(MODULE_EXTENSION ".obf")
   set(OB_MODULE_PATH "${_OB_DATADIR}")
   set(OB_SHARED_PTR_HEADER "memory")
   set(OB_SHARED_PTR_IMPLEMENTATION "std::shared_ptr")
   set(HAVE_SHARED_POINTER 1)
   set(HAVE_STD_SHARED_PTR 1)
   set(HAVE_GCC_VISIBILITY 1)
   set(HAVE_CONIO_H 1)
   set(HAVE_SYS_TIME_H 1)
   set(HAVE_TIME_H 1)
   set(HAVE_SSTREAM 1)
   set(HAVE_CLOCK_T 1)
   set(HAVE_RINT 1)
   set(HAVE_SNPRINTF 1)
   set(HAVE_STRCASECMP 1)
   set(HAVE_STRNCASECMP 1)
   file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/modules/openbabel/include/openbabel)
   configure_file(
      ${CMAKE_CURRENT_SOURCE_DIR}/modules/openbabel/src/config.h.cmake
      ${CMAKE_CURRENT_BINARY_DIR}/modules/openbabel/include/openbabel/babelconfig.h)

   #----------------------------------------------------------------
   # 关键: 让 OpenBabel 子模块的 BUILD_SHARED 跟随顶层 BUILD_SHARED_LIBS。
   #
   # OpenBabel 用的是自己的私有开关 BUILD_SHARED (modules/openbabel/
   # CMakeLists.txt:105, 默认 ON), 与 CMake 标准的 BUILD_SHARED_LIBS
   # 互不相干。若不显式同步, 在顶层 BUILD_SHARED_LIBS=OFF (静态构建)
   # 的情况下 OpenBabel 仍按 BUILD_SHARED=ON 配置, 导致:
   #   - 定义 -DUSING_DYNAMIC_LIBS, 插件不再内联进主库,
   #     而是编成独立 MODULE (.obf/.so) 期望运行时从磁盘加载;
   #   - 运行期 OBPlugin::LoadAllPlugins() (src/plugin.cpp:46) 进入
   #     #if defined(USING_DYNAMIC_LIBS) 分支, 在 BABEL_LIBDIR 下找
   #     插件文件, 找不到即报:
   #       "Unable to find OpenBabel plugins.
   #        Try setting the BABEL_LIBDIR environment variable."
   #   - 进而 FindForceField("UFF") 返回 NULL -> 能量计算报
   #     "Failed to load force field: UFF"。
   #
   # 必须在 add_subdirectory(modules/openbabel) 之前设置; 用 FORCE 保证
   # 覆盖已被缓存/已定义的 option 值。
   #----------------------------------------------------------------
   if (NOT BUILD_SHARED_LIBS)
      set(BUILD_SHARED OFF CACHE BOOL "OpenBabel: build shared library + dynamic plugins" FORCE)
   else()
      set(BUILD_SHARED ON CACHE BOOL "OpenBabel: build shared library + dynamic plugins" FORCE)
   endif()

   # 静态构建 OpenBabel 时, 其捆绑的 maeparser (schrodinger::mae 命名空间) 默认
   # 按"DLL 导入"处理, 会在 MinGW 链接期产生 undefined reference:
   #   __imp__ZN11schrodinger3mae6Reader... (MAE 格式)
   # 原因: maeparser 的 MaeParserConfig.hpp 用 STATIC_MAEPARSER 控制导出宏:
   #     #ifndef STATIC_MAEPARSER
   #       #define EXPORT_MAEPARSER __declspec(dllimport)   // 消费者侧
   #     #else
   #       #define EXPORT_MAEPARSER                          // 静态: 空
   #     #endif
   # 而 maeparser/CMakeLists.txt 只做了
   #   target_compile_definitions(maeparser PRIVATE "STATIC_MAEPARSER")
   # —— PRIVATE 不会传播给消费者(openbabel 的 maeformat.cpp), 故消费者
   # 仍按 DLL 导入展开。必须在消费者侧补上 STATIC_MAEPARSER。
   # 注意: 宏名是 STATIC_MAEPARSER, 不是 MAEPARSER_STATIC_DEFINE
   # (后者是 GenerateExportHeader 的命名习惯, 对 maeparser v1.2.3 无效)。
   # 必须在 add_subdirectory 之前设置, 让 openbabel/maeparser 子目录继承该宏。
   #
   # coordgen 有完全相同的缺陷(其 CoordgenConfig.hpp 用 STATIC_COORDGEN 控制
   # 导出宏, 而 coordgenlibs/CMakeLists.txt 同样用 PRIVATE 定义不传播)。
   # 当前 WITH_COORDGEN 在静态构建下默认为 OFF、且 openbabel 源码不引用
   # coordgen 符号, 故不触发; 此处一并预防, 以便将来打开该选项时不会踩坑。
   if (NOT BUILD_SHARED_LIBS)
      add_compile_definitions(STATIC_MAEPARSER STATIC_COORDGEN)
   endif()

   add_subdirectory(modules/openbabel  EXCLUDE_FROM_ALL)

   #----------------------------------------------------------------
   # 补齐 OpenBabel 静态构建漏编的"格式附加源" (per-format _additional_sources)
   #
   # OpenBabel 的 src/formats/formats.cmake 可为每个格式定义可选的
   # ${format}_additional_sources, 里面放着该格式真正需要的实现, 例如:
   #   asciiformat  -> ../depict/asciipainter.cpp   (OpenBabel::ASCIIPainter)
   #   painterformat-> ../depict/commandpainter.cpp (OpenBabel::CommandPainter)
   #   wlnformat    -> wln-nextmove.cpp             (NMReadWLN)
   # 然而 OpenBabel 的 src/CMakeLists.txt 在静态构建分支 (BUILD_SHARED=OFF) 只做了
   #   foreach(format ${formats})  set(openbabel_srcs ... formats/${format}.cpp)
   # 完全忽略了 ${format}_additional_sources, 导致这些 .cpp 根本没被编进
   # libopenbabel.a。最终链 IQmol.exe 时报 undefined reference:
   #   ASCIIPainter / CommandPainter / NMReadWLN
   # (共享构建走 add_subdirectory(formats) 会正确带上 additional_sources, 故仅
   #  静态构建踩坑。)
   #
   # 修复: 在 IQmol 侧把漏掉的源显式补进 openbabel 目标, 随其一起编入静态库。
   # 只补"始终启用"的格式所需源; png2format 的 cairopainter.cpp 仅在 CAIRO_FOUND
   # 时才需要(本构建默认不带 Cairo, 故此处不加); inchiformat 的附加源由 openbabel
   # 自己在 WITH_STATIC_INCHI 分支处理, 亦不在此列。
   #
   # 文件存在性用 EXISTS 探一下, 避免将来 openbabel 升版改了路径导致 configure 失败。
   #
   # 注意: 仅对内嵌的"本地" openbabel 目标补源。Linux 等平台走
   # find_package(OpenBabel3) 时 openbabel 是 IMPORTED 目标(指向系统 .so),
   # CMake 禁止给 IMPORTED 目标追加源(target_sources 会直接报错), 且系统库本身
   # 已编好, 无需也不可能补源。故这里先排除 IMPORTED 场景。
   #
   # 去重: 这些附加源也可能已由 openbabel 自身(或其 CMakeLists 被外部补丁改过,
   # 例如 scripts/build_windows.sh 步骤 2d 给 openbabel/src/CMakeLists.txt 打补丁)
   # 编入目标。若目标 SOURCES 里已含同名文件, 此处跳过, 避免同一 .cpp 被加两次
   # 触发重复定义/链接错误。
   #----------------------------------------------------------------
   set(_ob_is_imported FALSE)
   if (TARGET openbabel)
      get_target_property(_ob_is_imported openbabel IMPORTED)
   endif()
   if (TARGET openbabel AND NOT _ob_is_imported)
      # 收集 openbabel 目标当前已含的源文件基名, 用于去重
      set(_ob_existing_names "")
      get_target_property(_ob_existing_srcs openbabel SOURCES)
      if (_ob_existing_srcs)
         foreach(_s ${_ob_existing_srcs})
            get_filename_component(_s_name "${_s}" NAME)
            list(APPEND _ob_existing_names "${_s_name}")
         endforeach()
      endif()
      set(_ob_missing_srcs "")
      foreach(_ob_src
            modules/openbabel/src/depict/asciipainter.cpp
            modules/openbabel/src/depict/commandpainter.cpp
            modules/openbabel/src/formats/wln-nextmove.cpp)
         if (EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${_ob_src}")
            get_filename_component(_ob_base "${_ob_src}" NAME)
            list(FIND _ob_existing_names "${_ob_base}" _ob_idx)
            if (_ob_idx EQUAL -1)
               list(APPEND _ob_missing_srcs "${_ob_src}")
            else()
               message(STATUS "[IQmol] 附加源 ${_ob_base} 已由 openbabel 自身编入, 跳过重复添加")
            endif()
         endif()
      endforeach()
      if (_ob_missing_srcs)
         target_sources(openbabel PRIVATE ${_ob_missing_srcs})
         message(STATUS "[IQmol] 已为静态 openbabel 补齐漏编的附加源: ${_ob_missing_srcs}")
      else()
         message(STATUS "[IQmol] 无需补齐 openbabel 附加源(均已就位或由 openbabel 自身编入)")
      endif()
      unset(_ob_missing_srcs)
      unset(_ob_existing_srcs)
      unset(_ob_existing_names)
   endif()
   unset(_ob_is_imported)
endif()

# libarchive (静态库 target: archive_static; bcrypt 为 Windows 下的 SHA 依赖。
# 上游 WIN32 分支原引用不存在的 modules/zstd 导致 configure 必崩, 已移除;
# zstd 为 libarchive 可选依赖, 缺失时自动跳过, IQmol 无需 .zst 支持;
# MinGW 下 libarchive 自带 CHECK_ICONV 探测不可靠, 生成的 config.h 把 ICONV_CONST
# 定义成空; 而 OpenBabel 自带 iconv.h 声明 iconv 形参 inbuf 为 const char**,
# 导致 archive_string.c:2092 iconv(cd, &itp,...) 实参 char** 与 const char** 冲突。
# 命令行 -DICONV_CONST=const 会被 config.h 后定义反覆盖(实测 redefined 仍报错),
# 故在 configure 阶段对子模块源做幂等修补(见 WIN32 分支), 调用处实参转型为
# (const char **)&itp, 与 OpenBabel iconv.h 的 const char** 声明对齐)
if (LINUX)
   find_package(LibArchive REQUIRED)
   set(ARCHIVE_LIBRARY ${LibArchive_LIBRARIES})
elseif (APPLE)
   find_package(ZSTD REQUIRED)
   if (EXISTS /opt/homebrew/lib/)
      link_directories(/opt/homebrew/lib/)
   endif()
   include_directories(modules/libarchive/libarchive)
   add_subdirectory(modules/libarchive EXCLUDE_FROM_ALL)
   set(ARCHIVE_LIBRARY archive_static)
else (WIN32)
   include_directories(modules/libarchive)
   # 修复 MinGW 下 libarchive 调用 libiconv 的签名冲突: 对子模块源做幂等修补。
   # OpenBabel 自带 iconv.h 声明 inbuf 为 const char**, 而 libarchive 的 config.h
   # 把 ICONV_CONST 定义成空, 使 archive_string.c:2092 的 &itp 为 char** 不匹配。
   # 此处把调用处实参转型为 (const char **)&itp (已修补则跳过, 避免重复),
   # 它在任意 ICONV_CONST 取值下都合法, 比强行覆盖宏更鲁棒。
   set(_LA_STRING "${CMAKE_SOURCE_DIR}/modules/libarchive/libarchive/archive_string.c")
   if (EXISTS "${_LA_STRING}")
      file(READ "${_LA_STRING}" _LA_SRC)
      string(FIND "${_LA_SRC}" "iconv(cd, (const char **)&itp," _LA_DONE)
      if (_LA_DONE EQUAL -1)
         string(REPLACE "iconv(cd, &itp, &remaining, &outp, &avail)"
                         "iconv(cd, (const char **)&itp, &remaining, &outp, &avail)"
                         _LA_SRC "${_LA_SRC}")
         file(WRITE "${_LA_STRING}" "${_LA_SRC}")
         message(STATUS "[IQmol] 已修补 libarchive archive_string.c: iconv 实参 &itp -> (const char**)&itp")
      endif()
   endif()
   add_subdirectory(modules/libarchive EXCLUDE_FROM_ALL)
   set(ARCHIVE_LIBRARY archive_static bcrypt)
endif()

   # OpenSSL: Network 的 SshConnection.C 直接调用 MD5() 等 OpenSSL 函数,
   # 但 Network 以 PRIVATE 方式链接 OpenSSL::SSL —— 静态库的 PRIVATE 依赖
   # 不会传播到最终可执行文件, 故顶层必须显式补链, 否则链接期报
   # undefined reference to `MD5' / `__imp_MD5'。
   OpenSSL::SSL
   OpenSSL::Crypto
   ${ZLIB_LIBRARIES}
   ${FORTRAN_LIBRARIES}
   ${ARCHIVE_LIBRARY}
)

#-------------------------------------------------------------------
# 静态链接 OpenBabel 插件整体保留 (Windows/MinGW 必需):
#
#   OpenBabel 的插件(力场/格式/描述符/指纹等)不是通过显式函数调用注册的,
#   而是靠源文件里的全局对象在 main() 之前自注册, 例如:
#       forcefields/forcefielduff.cpp:579
#       OBForceFieldUFF theForceFieldUFF("UFF", true);
#   构造函数把实例塞进 OBForceField::Map()。
#
#   BUILD_SHARED_LIBS OFF 时这些 .cpp 被编成静态库 openbabel.a 的成员。
#   GNU ld 只按"符号引用"提取静态库成员: 若某个 .o 里的符号没有任何地方
#   引用(纯全局构造副作用), 该 .o 会被整个丢弃, 自注册随之失效。
#   症状: 运行时 FindForceField("UFF") 返回 NULL, IQmol 弹出
#         "Failed to load force field: UFF", 但编译链接期毫无报错。
#
#   Linux 上通常因动态链接或系统包展开而恰好可用, Windows 静态构建必踩。
#
#   关键: 不能用裸 -Wl,--whole-archive openbabel -Wl,--no-whole-archive ——
#   CMake 在生成 link.txt 时会把"链接器旗标"与"库名"拆开重新排序, 结果
#   --whole-archive / --no-whole-archive 被错误地套在 IQmol 自己的 objects.a
#   上, 而 openbabel 被塞进 linkLibs.rsp 做普通链接, 自注册成员照样被丢弃。
#   修复: 用 CMake 官方封装 $<LINK_LIBRARY:WHOLE_ARCHIVE,openbabel>, 由 CMake
#   保证 --whole-archive 紧贴 openbabel 路径, 不惧排序。要求 cmake >= 3.24
#   (本项目 build_windows.sh 自带 3.31.6, 满足)。
#-------------------------------------------------------------------
#-------------------------------------------------------------------
# $<LINK_LIBRARY:WHOLE_ARCHIVE,...> 需要 CMake >= 3.24。
# 本项目 cmake_minimum_required 为 3.14.4, 低于该版本时该生成器表达式
# 会直接报错(不是被忽略), 故这里先探测一次, 低版本自动降级为裸旗标,
# 避免老环境上"配置期就挂掉"。
if (CMAKE_VERSION VERSION_GREATER_EQUAL 3.24)
   set(IQMOL_HAVE_LINK_LIBRARY TRUE)
else()
   set(IQMOL_HAVE_LINK_LIBRARY FALSE)
   message(STATUS
      "[IQmol] CMake ${CMAKE_VERSION} < 3.24, WHOLE_ARCHIVE 将降级为裸旗标; "
      "静态构建下力场插件自注册可能失效, 建议升级 CMake")
endif()


if (WIN32 AND NOT BUILD_SHARED_LIBS)
   if (TARGET openbabel)
      # 用 CMake 官方封装 $<LINK_LIBRARY:WHOLE_ARCHIVE,openbabel> 而非裸
      # -Wl,--whole-archive: 裸旗标会被 CMake 在生成 link.txt 时与库名拆开重排,
      # 错误地套到 IQmol 自己的 objects.a 上。官方封装保证 --whole-archive 紧贴
      # openbabel 路径, 并保留目标依赖跟踪(无需手写 $<TARGET_FILE:...> 路径)。
      if (IQMOL_HAVE_LINK_LIBRARY)
         target_link_libraries (${targetName}
            $<LINK_LIBRARY:WHOLE_ARCHIVE,openbabel>
         )
      else()
         # 低版本 CMake 降级：裸旗标顺序不保证, 但至少不中断配置
         target_link_libraries (${targetName}
            -Wl,--whole-archive openbabel -Wl,--no-whole-archive
         )
      endif()
      # 双保险：显式声明依赖, 确保 openbabel 在链接前被构建
      # (openbabel 经 EXCLUDE_FROM_ALL 引入, 不属于默认 all 目标)
      add_dependencies(${targetName} openbabel)
      message(STATUS "[IQmol] 已对 openbabel 启用 --whole-archive (保证静态插件自注册不被丢弃)")
   else()
      target_link_libraries (${targetName} openbabel)
      message(WARNING "[IQmol] 未找到 openbabel 目标, 直接按名称链接")
   endif()
else()
   # 非 Windows 分支。
   #
   # 这里要区分两种 openbabel：
   #   (a) 嵌入式静态构建 —— openbabel 是本地源码编译出的 STATIC 库目标。
   #       其插件(力场/格式/描述符)靠全局对象自注册, 静态库成员会因"无符号
   #       引用"被链接器丢弃, 必须整体保留。
   #   (b) 系统包构建 —— Linux 下 find_package(OpenBabel3) 返回的是 IMPORTED
   #       目标, 指向 libopenbabel.so。它本身是共享库, 根本不存在"成员被丢弃"
   #       的问题, 却在实体上又经依赖链被链接了一次。此时再叠加 WHOLE_ARCHIVE
   #       会让 CMake 报错：
   #         Impossible to link target 'IQmol' because the link item 'openbabel'
   #         ... has already occurred with the feature 'WHOLE_ARCHIVE'
   #       因为同一链接项不能同时以"无特性"和"WHOLE_ARCHIVE 特性"出现。
   # 故仅对 (a) 启用 WHOLE_ARCHIVE; (b) 按普通链接处理。
   set(_iqmol_ob_is_imported FALSE)
   if (TARGET openbabel)
      get_target_property(_iqmol_ob_is_imported openbabel IMPORTED)
   endif()

   if (NOT BUILD_SHARED_LIBS AND TARGET openbabel AND NOT _iqmol_ob_is_imported)
      if (IQMOL_HAVE_LINK_LIBRARY)
         target_link_libraries (${targetName}
            $<LINK_LIBRARY:WHOLE_ARCHIVE,openbabel>
         )
      else()
         target_link_libraries (${targetName}
            -Wl,--whole-archive openbabel -Wl,--no-whole-archive
         )
      endif()
      add_dependencies(${targetName} openbabel)
      message(STATUS "[IQmol] 已对嵌入式静态 openbabel 启用 --whole-archive")
   else()
      if (NOT TARGET openbabel)
         # 没有任何 openbabel 目标：按名称链接, 交给链接器解析(系统库场景)
         target_link_libraries (${targetName} openbabel)
         message(STATUS "[IQmol] openbabel 按库名链接 (系统包)")
      elseif (_iqmol_ob_is_imported)
         # IMPORTED(系统共享库): 不重复链接, 沿用依赖链中的那一份。
         message(STATUS "[IQmol] openbabel 为系统共享库, 已由依赖链链接, 跳过 WHOLE_ARCHIVE")
      else()
         target_link_libraries (${targetName} openbabel)
      endif()
   endif()
   unset(_iqmol_ob_is_imported)
endif()

# maeparser 是 OpenBabel 捆绑的独立静态库(MAE 格式: schrodinger::mae 命名空间),
# 其实现符号同样可能被链接器按引用丢弃, 且需与上面的 STATIC_MAEPARSER 宏
# 配合(消除 __imp_ DLL 导入前缀)。若它是独立 CMake 目标则一并整体保留;
# 否则由 openbabel 的 PRIVATE 依赖传递链入(此时已被 openbabel 的 WHOLE_ARCHIVE
# 覆盖, 无需重复)。
if (TARGET maeparser)
   if (IQMOL_HAVE_LINK_LIBRARY)
      target_link_libraries (${targetName}
         $<LINK_LIBRARY:WHOLE_ARCHIVE,maeparser>
      )
   else()
      target_link_libraries (${targetName}
         -Wl,--whole-archive maeparser -Wl,--no-whole-archive
      )
   endif()
   add_dependencies(${targetName} maeparser)
   message(STATUS "[IQmol] 已对 maeparser 启用 --whole-archive")
endif()

if (WIN32)
   target_link_libraries (${targetName} ws2_32)
endif ()
